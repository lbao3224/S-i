Đổi Mật Khẩu

Cập nhật mật khẩu tài khoản Sổ Ơi

Mã SQL Schema Supabase

Chạy trong Supabase Dashboard -> SQL Editor

-- 1. BẢNG MẶT HÀNG (sooi_products)
CREATE TABLE IF NOT EXISTS public.sooi_products (
    id TEXT NOT NULL,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    category TEXT DEFAULT 'Khác',
    price NUMERIC DEFAULT 0,
    cost NUMERIC DEFAULT 0,
    stock INTEGER DEFAULT 0,
    "desc" TEXT DEFAULT '',
    img TEXT DEFAULT '',
    updated_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()),
    PRIMARY KEY (id, user_id)
);

-- 2. BẢNG KHÁCH HÀNG & CÔNG NỢ (sooi_customers)
CREATE TABLE IF NOT EXISTS public.sooi_customers (
    id TEXT NOT NULL,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    phone TEXT DEFAULT '',
    debt NUMERIC DEFAULT 0,
    memo TEXT DEFAULT '',
    updated_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()),
    PRIMARY KEY (id, user_id)
);

-- 3. BẢNG SỔ THU CHI (sooi_cash)
CREATE TABLE IF NOT EXISTS public.sooi_cash (
    id TEXT NOT NULL,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    type TEXT NOT NULL CHECK (type IN ('thu', 'chi')),
    amount NUMERIC DEFAULT 0,
    memo TEXT DEFAULT '',
    category TEXT DEFAULT 'Khác',
    date TEXT DEFAULT '',
    time TEXT DEFAULT '',
    method TEXT DEFAULT 'Tiền mặt',
    updated_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()),
    PRIMARY KEY (id, user_id)
);

-- 4. BẢNG ĐƠN HÀNG (sooi_orders)
CREATE TABLE IF NOT EXISTS public.sooi_orders (
    id TEXT NOT NULL,
    user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
    items JSONB DEFAULT '[]'::jsonb,
    total NUMERIC DEFAULT 0,
    cost NUMERIC DEFAULT 0,
    method TEXT DEFAULT 'Tiền mặt',
    customer_name TEXT DEFAULT '',
    date TEXT DEFAULT '',
    time TEXT DEFAULT '',
    created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()),
    PRIMARY KEY (id, user_id)
);

-- 5. BẢNG CÀI ĐẶT CỬA HÀNG (sooi_settings)
CREATE TABLE IF NOT EXISTS public.sooi_settings (
    user_id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    shop_name TEXT DEFAULT 'Cà Phê Sáng Sổ Ơi',
    bank_bin TEXT DEFAULT '970436',
    bank_name TEXT DEFAULT 'Vietcombank',
    bank_acc TEXT DEFAULT '',
    account_name TEXT DEFAULT '',
    block_negative_stock BOOLEAN DEFAULT false,
    updated_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now())
);

-- BẬT ROW LEVEL SECURITY (RLS)
ALTER TABLE public.sooi_products ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.sooi_customers ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.sooi_cash ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.sooi_orders ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.sooi_settings ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Users manage own products" ON public.sooi_products FOR ALL USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id);
CREATE POLICY "Users manage own customers" ON public.sooi_customers FOR ALL USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id);
CREATE POLICY "Users manage own cash" ON public.sooi_cash FOR ALL USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id);
CREATE POLICY "Users manage own orders" ON public.sooi_orders FOR ALL USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id);
CREATE POLICY "Users manage own settings" ON public.sooi_settings FOR ALL USING (auth.uid() = user_id) WITH CHECK (auth.uid() = user_id);
